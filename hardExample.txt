// Base rivers (flows are scaled by rainfall factor at runtime)
river Styx = 4.0x;
river Acheron = 5.0x;
river Lethe = 3.0x;
river Cocytus = 2.0x;
river Phlegethon = 1.5x;   
river Oceanus = 6.0x;   

// --- Dam definitions -------------------------------------------------
// Dams subtract a fixed flow loss when their gate is "open" (gate > threshold).

dam Z(inflow, loss, gate) {
  if (inflow > 15) {
    return inflow/2;   
  }
  return inflow;
}

dam H(inflow, loss, gate) {
  if (gate == 1) {
    return inflow !~ loss;
  }
  if (rainfall > 5){
    return inflow !~ loss;
  }
  return inflow;
}

// --- Local fixed losses expressed as flows ---------------------------
river lossZ = 0.6x;
river lossH = 0.8x;

// --- Gate settings (plain numbers). Adjust these to open/close dams --
// Idea is that workers can manually open or close these Gates

river gateZ = 0;           // 0 = close
river gateH = 1;           // 1 = open

// --- Intermediate junctions -----------------------------------------
river Erebus     = 0.0x;   // node1: Styx through Z, then join with Acheron
river Hypnos     = 0.0x;   // node2: Lethe through H, then join with Cocytus
river Nexus      = 0.0x;   // final junction of node1 and node2


// --- Network topology -----------------------------------------------

Erebus <- Z(Styx, lossZ, gateZ) ~~ Acheron;
Hypnos <- H(Lethe, lossH, gateH) ~~ Cocytus;
Nexus <- Erebus ~~ Hypnos;

// Oceanus collects its base, Phlegethon, and everything arriving at Nexus
Oceanus <- Oceanus ~~ Phlegethon ~~ Nexus;

// --- Reporting -------------------------------------------------------
print("");

print("=== Base Rivers (m³/s depends on runtime rainfall) ===");
print(Styx);

print(Phlegethon);  
print("");

print("=== Dam Loss Settings ===");
print(lossZ);
print(lossH);
print("");

print("=== Gates (open/close) ===");
print(gateZ);
print(gateH);
print("");

print("=== Intermediate Nodes ===");
print(Erebus);
print(Hypnos);
print(Nexus);
print("");

print("=== Final Outflow (Oceanus) ===");
print(Oceanus);
print("");
